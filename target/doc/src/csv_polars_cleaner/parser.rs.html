6/// Return Polars DataFrame using box logic, preserving internal newlines
7pub fn parse_boxed_data(raw: &str, delimiter: u8) -> Result<DataFrame> {
8    // Normalize line endings and split into lines
9    let normalized = crate::preprocessor::normalize_line_endings(raw);
10    let mut lines: Vec<&str> = normalized.lines().collect();
11
12    // Remove empty lines
13    lines.retain(|l| !l.trim().is_empty());
14
15    // Detect the data bounds, skipping comments and empty lines
16    let (start, end) = crate::detector::detect_data_bounds(&lines, delimiter as char)
17        .context("Could not detect valid data box")?;
18
19    // Preprocess the boxed data: trim whitespace from each field and repair malformed rows
20    let header = lines[start].split(delimiter as char)
21        .map(|s| s.trim())
22        .collect::<Vec<_>>();
23    let num_cols = header.len();
24
25    let mut processed_lines = vec![header.join(&String::from(delimiter as char))];
26    for line in &lines[start+1..=end] {
27        if line.trim().is_empty() || line.trim().starts_with('#') {
28            continue;
29        }
30        let mut fields: Vec<String> = line.split(delimiter as char)
31            .map(|s| s.trim().to_string())
32            .collect();
33        // If not enough fields, try splitting by whitespace
34        if fields.len() < num_cols {
35            let ws_fields: Vec<String> = line.split_whitespace().map(|s| s.trim().to_string()).collect();
36            if ws_fields.len() == num_cols {
37                fields = ws_fields;
38            } else if ws_fields.len() > num_cols && num_cols > 1 {
39                // Merge trailing fields into the last column
40                let mut merged: Vec<String> = ws_fields[..num_cols-1].to_vec();
41                merged.push(ws_fields[num_cols-1..].join(" "));
42                fields = merged;
43            }
44        }
45        // Pad with empty if still not enough fields
46        while fields.len() < num_cols {
47            fields.push(String::new());
48        }
49        processed_lines.push(fields.join(&String::from(delimiter as char)));
50    }
51
52    // Join the processed lines
53    let boxed = processed_lines.join("\n");
54
55    // Create a Cursor from the string bytes
56    let cursor = Cursor::new(boxed.as_bytes());
57
58    // Configure CSV parsing options
59    let parse_options = CsvParseOptions::default()
60        .with_separator(delimiter)
61        .with_try_parse_dates(false)
62        .with_missing_is_null(true)
63        .with_comment_prefix(Some("#"));
64
65    let read_options = CsvReadOptions::default()
66        .with_parse_options(parse_options)
67        .with_has_header(true)
68        .with_skip_rows(0);
69
70    let df = read_options
71        .into_reader_with_file_handle(cursor)
72        .finish()?;
73
74    // Convert all columns to String (string)
75    let df = df.get_columns()
76        .iter()
77        .map(|s| s.cast(&DataType::String))
78        .collect::<polars::prelude::PolarsResult<Vec<_>>>()?;
79    let df = DataFrame::new(df)?;
80
81    Ok(df)
82}

97    #[test]
98    fn test_mixed_line_endings() {
99        // Create a malformed CSV with mixed line endings and metadata
100        let input = r#"# This is a comment
101# Another comment
102# Metadata: version=1.0
103
104name,age,city
105Alice,30,New York
106Bob,25,London
107Charlie,35,Paris
108
109# End of file"#;
110
111        let result = parse_boxed_data(input, b',');
112        assert!(result.is_ok(), "Failed to parse CSV with mixed line endings");
113        
114        let df = result.unwrap();
115        let expected = create_expected_df();
116        assert!(df.equals(&expected), "DataFrame content mismatch");
117    }
